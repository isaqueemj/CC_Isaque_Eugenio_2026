CREATE DATABASE IF NOT EXISTS Sistema;
USE Sistema;
-- =========================
-- PAÍSES
-- =========================
CREATE TABLE IF NOT EXISTS paises (
    id_pais INT AUTO_INCREMENT PRIMARY KEY,
    pais VARCHAR(54) NOT NULL,
    sigla CHAR(3) UNIQUE NOT NULL,
    ddi CHAR(5) NOT NULL,
    moeda CHAR(3),
    nacionalidade VARCHAR(30),
    codigo_ibge INT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- =========================
-- ESTADOS
-- =========================
CREATE TABLE IF NOT EXISTS estados (
    id_estado INT AUTO_INCREMENT PRIMARY KEY,
    id_pais INT NOT NULL,
    estado VARCHAR(50) NOT NULL,
    uf CHAR(2) NOT NULL,
    codigo_ibge INT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_estado_uf_pais (uf, id_pais),
    CONSTRAINT fk_estado_pais FOREIGN KEY (id_pais) REFERENCES paises(id_pais)
);

-- =========================
-- CIDADES
-- =========================
CREATE TABLE IF NOT EXISTS cidades (
    id_cidade INT AUTO_INCREMENT PRIMARY KEY,
    id_estado INT NOT NULL,
    cidade VARCHAR(60) NOT NULL,
    ddd VARCHAR(3),
    codigo_ibge INT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_cidade_estado FOREIGN KEY (id_estado) REFERENCES estados(id_estado)
);

-- =========================
-- FORNECEDORES (COM ENDEREÇO EMBUTIDO)
-- =========================
CREATE TABLE IF NOT EXISTS fornecedores (
    id_fornecedor INT AUTO_INCREMENT PRIMARY KEY,
    fornecedor VARCHAR(50) NOT NULL,
    categoria VARCHAR(50),

    -- contato / info embutida
    email VARCHAR(80),
    cpf_cnpj CHAR(14),
    telefone VARCHAR(20),
    celular VARCHAR(20),
    ie VARCHAR(14),
    tipo_pessoa CHAR(1),
    rg CHAR(11),
    observacao TEXT,

    -- endereço embutido
    bairro VARCHAR(60),
    logradouro VARCHAR(100),
    numero VARCHAR(10),
    complemento VARCHAR(100),
    cep CHAR(8),

    -- localização
    id_cidade INT NOT NULL,

    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_fornecedor_cidade FOREIGN KEY (id_cidade) REFERENCES cidades(id_cidade)
);

-- =========================
-- CLIENTES (COM ENDEREÇO EMBUTIDO)
-- =========================
CREATE TABLE IF NOT EXISTS clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    cliente VARCHAR(50) NOT NULL,
    apelido VARCHAR(50),
    tipo VARCHAR(20),

    -- contato / info embutida
    email VARCHAR(80),
    cpf_cnpj CHAR(14),
    telefone VARCHAR(20),
    celular VARCHAR(20),
    ie VARCHAR(14),
    tipo_pessoa CHAR(1),
    rg CHAR(11),
    observacao TEXT,

    -- endereço embutido
    bairro VARCHAR(60),
    logradouro VARCHAR(100),
    numero VARCHAR(10),
    complemento VARCHAR(100),
    cep CHAR(8),

    -- localização
    id_cidade INT NOT NULL,

    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_cliente_cidade FOREIGN KEY (id_cidade) REFERENCES cidades(id_cidade)
);

-- =========================
-- MARCAS
-- =========================
CREATE TABLE IF NOT EXISTS marcas (
    id_marca INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    slug VARCHAR(100) UNIQUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- =========================
-- MODELOS
-- =========================
CREATE TABLE IF NOT EXISTS modelos (
    id_modelo INT AUTO_INCREMENT PRIMARY KEY,
    id_marca INT NOT NULL,
    modelo VARCHAR(30) NOT NULL,
    descricao TEXT,
    ano_fabricacao YEAR,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_modelo_marca FOREIGN KEY (id_marca) REFERENCES marcas(id_marca)
);

-- =========================
-- VEÍCULOS
-- =========================
CREATE TABLE IF NOT EXISTS veiculos (
    id_veiculo INT AUTO_INCREMENT PRIMARY KEY,
    veiculo VARCHAR(100),
    placa VARCHAR(10) UNIQUE NOT NULL,
    renavam VARCHAR(20) UNIQUE,
    chassi VARCHAR(30) UNIQUE,
    cor VARCHAR(30),
    ano_fabricacao YEAR,
    ano_modelo YEAR,
    combustivel VARCHAR(20),
    quilometragem INT DEFAULT 0,
    fk_modelo INT NOT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_veiculo_modelo FOREIGN KEY (fk_modelo) REFERENCES modelos(id_modelo)
);

-- =========================
-- PRODUTOS
-- =========================
CREATE TABLE IF NOT EXISTS produtos (
    id_produto INT AUTO_INCREMENT PRIMARY KEY,
    produto VARCHAR(50) NOT NULL,
    sku VARCHAR(30),
    categoria VARCHAR(50),
    unidade VARCHAR(3) DEFAULT 'UN',
    preco_custo DECIMAL(10,2) DEFAULT 0,
    preco_venda DECIMAL(10,2) DEFAULT 0,
    saldo DECIMAL(10,2) DEFAULT 0,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- =========================
-- CONDIÇÕES DE PAGAMENTO
-- =========================
CREATE TABLE IF NOT EXISTS condicoes_pagamento (
    id_condicao_pagamento INT AUTO_INCREMENT PRIMARY KEY,
    condicao_pagamento VARCHAR(60) NOT NULL,
    quantidade_parcelas INT DEFAULT 1,
    intervalo_dias INT DEFAULT 30,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- =========================
-- FORMAS DE PAGAMENTO
-- =========================
CREATE TABLE IF NOT EXISTS formas_pagamento (
    id_forma_pagamento INT AUTO_INCREMENT PRIMARY KEY,
    forma_pagamento VARCHAR(60) NOT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- =========================
-- TRANSPORTADORES
-- =========================
CREATE TABLE IF NOT EXISTS transportadores (
    id_transportador INT AUTO_INCREMENT PRIMARY KEY,
    transportador VARCHAR(50) NOT NULL,
    tipo_frete VARCHAR(20),

    -- endereço + contato embutido
    email VARCHAR(80),
    telefone VARCHAR(20),
    celular VARCHAR(20),
    cpf_cnpj CHAR(14),

    bairro VARCHAR(60),
    logradouro VARCHAR(100),
    numero VARCHAR(10),
    complemento VARCHAR(100),
    cep CHAR(8),

    id_cidade INT NOT NULL,

    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_transportador_cidade FOREIGN KEY (id_cidade) REFERENCES cidades(id_cidade)
);

CONTAS A PAGAR DEVE ESTAR VINCULADO A COMPRA QUE E DIFERENTE DE CONDICAO DE PAGAMENTO E FORMA DE PAGAMENTO

-- =========================
-- COMPRAS (CABEÇALHO)
-- =========================
CREATE TABLE IF NOT EXISTS compras (
    id_compra INT AUTO_INCREMENT PRIMARY KEY,
    id_fornecedor INT NOT NULL,
    id_condicao_pagamento INT NOT NULL,
    id_forma_pagamento INT NOT NULL,
    id_transportador INT NULL,
    
    numero_nota_fiscal VARCHAR(50),
    data_emissao DATE NOT NULL,
    data_chegada DATE,
    
    valor_frete DECIMAL(10,2) DEFAULT 0,
    valor_desconto DECIMAL(10,2) DEFAULT 0,
    valor_total DECIMAL(10,2) DEFAULT 0,
    observacao TEXT,
    
    status VARCHAR(20) DEFAULT 'PENDENTE', -- PENDENTE, FINALIZADA, CANCELADA
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_compra_fornecedor FOREIGN KEY (id_fornecedor) REFERENCES fornecedores(id_fornecedor),
    CONSTRAINT fk_compra_condicao FOREIGN KEY (id_condicao_pagamento) REFERENCES condicoes_pagamento(id_condicao_pagamento),
    CONSTRAINT fk_compra_forma FOREIGN KEY (id_forma_pagamento) REFERENCES formas_pagamento(id_forma_pagamento),
    CONSTRAINT fk_compra_transportador FOREIGN KEY (id_transportador) REFERENCES transportadores(id_transportador)
);

-- =========================
-- ITENS DA COMPRA
-- =========================
CREATE TABLE IF NOT EXISTS itens_compra (
    id_item_compra INT AUTO_INCREMENT PRIMARY KEY,
    id_compra INT NOT NULL,
    id_produto INT NOT NULL,
    
    quantidade DECIMAL(10,3) NOT NULL,
    valor_unitario DECIMAL(10,2) NOT NULL,
    valor_total DECIMAL(10,2) NOT NULL,
    
    CONSTRAINT fk_item_compra FOREIGN KEY (id_compra) REFERENCES compras(id_compra) ON DELETE CASCADE,
    CONSTRAINT fk_item_produto FOREIGN KEY (id_produto) REFERENCES produtos(id_produto)
);

-- =========================
-- CONTAS A PAGAR
-- =========================
CREATE TABLE IF NOT EXISTS contas_pagar (
    id_conta_pagar INT AUTO_INCREMENT PRIMARY KEY,
    id_compra INT NOT NULL,
    id_fornecedor INT NOT NULL, -- Redundância intencional para facilitar relatórios rápidos
    id_forma_pagamento INT NOT NULL, 
    
    numero_parcela INT DEFAULT 1,
    descricao VARCHAR(100),
    
    data_vencimento DATE NOT NULL,
    data_pagamento DATE NULL,
    
    valor_parcela DECIMAL(10,2) NOT NULL,
    valor_pago DECIMAL(10,2) DEFAULT 0,
    valor_multa_juros DECIMAL(10,2) DEFAULT 0,
    valor_desconto DECIMAL(10,2) DEFAULT 0,
    
    status VARCHAR(20) DEFAULT 'ABERTA', -- ABERTA, PAGA, CANCELADA
    observacao TEXT,
    
    ativo BOOLEAN DEFAULT TRUE,
    data_cadastro DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_conta_compra FOREIGN KEY (id_compra) REFERENCES compras(id_compra) ON DELETE CASCADE,
    CONSTRAINT fk_conta_fornecedor FOREIGN KEY (id_fornecedor) REFERENCES fornecedores(id_fornecedor),
    CONSTRAINT fk_conta_forma_pgto FOREIGN KEY (id_forma_pagamento) REFERENCES formas_pagamento(id_forma_pagamento)
);
